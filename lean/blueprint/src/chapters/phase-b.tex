\chapter{Phase B --- Hopf algebra correspondence and the embedding theorem}

Phase~B bridges the geometric and algebraic sides of affine group schemes
and proves the embedding theorem stated in
Theorem~\ref{thm:embedding-statement}. The mathematical content is
classical (Waterhouse, Springer); the engineering work is to set up the
Lean infrastructure so the proof, when written, is a clean composition
of finite-generation of comodules, the regular representation, and the
Hopf algebra equivalence.

\section{The Spec functor on Hopf algebras}

\begin{definition}[Spec functor on Hopf algebras]\label{def:hopf-spec}\issue{9}
\uses{def:affine-group-scheme}
Let $R$ be a commutative ring. The functor
\[
  \operatorname{Spec} : (R\text{-CommAlg})^{\mathrm{op}} \to \mathsf{Over}(\Spec R)
\]
lifts to a functor on Hopf algebras and group objects:
\[
  \operatorname{hopfSpec} : \mathsf{Grp}\bigl((R\text{-CommAlg})^{\mathrm{op}}\bigr) \to \mathsf{Grp}\bigl(\mathsf{Over}(\Spec R)\bigr).
\]
Concretely, a commutative Hopf $R$-algebra is sent to its spectrum, which
inherits a group object structure from the Hopf structure via the
contravariant equivalence between commutative algebras and affine schemes.
\end{definition}

\begin{theorem}[\texorpdfstring{$\operatorname{hopfSpec}$}{hopfSpec} is fully faithful]
  \label{thm:hopf-spec-fully-faithful}\issue{9}
\uses{def:hopf-spec}
The functor $\operatorname{hopfSpec}$ is fully faithful.
\end{theorem}

\begin{theorem}[Essential image]\label{thm:hopf-spec-ess-image}\issue{9}
\uses{def:hopf-spec, def:affine-group-scheme}
The essential image of $\operatorname{hopfSpec}$ consists exactly of the
affine group schemes over $\Spec R$ --- equivalently, those group objects
$G \in \mathsf{Grp}(\mathsf{Over}(\Spec R))$ whose underlying scheme
$G.X$ is affine.
\end{theorem}

\section{Comodules and representations}

\begin{definition}[$G$-comodule, equivalently a representation]
  \label{def:G-representation}\issue{10}
\uses{def:hopf-spec, def:points}
Let $k$ be a field and $G$ an affine group scheme over $\Spec k$ with
coordinate Hopf algebra $A$. A $G$-representation on a $k$-vector space
$V$ is an $A$-comodule structure $\rho : V \to V \otimes_k A$ satisfying
coassociativity $(\mathrm{id}_V \otimes \Delta)\rho = (\rho \otimes \mathrm{id}_A)\rho$
and the counit axiom $(\mathrm{id}_V \otimes \varepsilon)\rho = \mathrm{id}_V$.
\end{definition}

\begin{theorem}[Representations versus group-scheme homomorphisms]
  \label{thm:repn-vs-hom}\issue{10}
\uses{def:G-representation, def:points}
For a finite-dimensional $k$-vector space $V$ and an affine group scheme
$G$ over $\Spec k$, the data of a $G$-representation on $V$ is naturally
equivalent to the data of a morphism of group schemes $G \to \GL(V)$.
\end{theorem}

\begin{example}[Trivial representation]\label{ex:trivial-repn}\issue{10}
\uses{def:G-representation}
For any $V$, the comodule structure $\rho(v) = v \otimes 1$ is a
$G$-representation, called the \emph{trivial representation}.
\end{example}

\begin{example}[Regular representation]\label{ex:regular-repn}\issue{10}
\uses{def:G-representation}
The coordinate Hopf algebra $A$ of $G$ is itself a $G$-representation on
the $k$-vector space $A$, with comodule structure
$\rho = \Delta : A \to A \otimes_k A$. This is the \emph{regular
representation}; it is generally infinite-dimensional.
\end{example}

\section{Finite generation of comodules}

\begin{lemma}[Finite-dimensional sub-comodule containing a given subspace]
  \label{lem:fg-comodule}\issue{11}
\uses{def:G-representation}
Let $A$ be a finitely generated commutative Hopf $k$-algebra and $V$ an
$A$-comodule. For every finite-dimensional $k$-subspace $V_0 \subseteq V$
there exists a finite-dimensional $G$-stable subspace
$V_0 \subseteq W \subseteq V$.
\end{lemma}

\begin{lemma}[The regular representation contains an algebra-generating sub-comodule]
  \label{lem:regular-generates}\issue{11}
\uses{lem:fg-comodule, ex:regular-repn}
Let $A$ be a finitely generated commutative Hopf $k$-algebra and pick a
finite set of $k$-algebra generators $a_1, \ldots, a_n \in A$. There
exists a finite-dimensional $G$-stable subspace $W \subseteq A$
containing each $a_i$. In particular, $W$ generates $A$ as a
$k$-algebra.
\end{lemma}

\section{The embedding theorem}

\begin{theorem}[Embedding theorem]\label{thm:embedding}\issue{12}
\uses{thm:hopf-spec-ess-image, thm:repn-vs-hom, lem:regular-generates}
Let $G$ be an algebraic group over a field $k$. Then $G$ admits a
faithful finite-dimensional representation. Equivalently, $G$ is
isomorphic, as a group scheme over $\Spec k$, to a closed subgroup
scheme of some $\GL_n$.
\end{theorem}
\begin{proof}
By Theorem~\ref{thm:hopf-spec-ess-image}, $G \simeq \operatorname{Spec} A$
for a finitely generated commutative Hopf $k$-algebra $A$. Apply
Lemma~\ref{lem:regular-generates} to obtain a finite-dimensional
$G$-stable subspace $W \subseteq A$ generating $A$ as a $k$-algebra. By
Theorem~\ref{thm:repn-vs-hom}, the $G$-action on $W$ gives a morphism
$\rho : G \to \GL(W)$. The pullback ring map
$\rho^* : \operatorname{Sym}(W^*) \otimes (\det)^{-1} \to A$ is
surjective because $W$ generates $A$. Hence $\rho$ is a closed immersion,
and $\rho$ is a faithful finite-dimensional representation.
\end{proof}

\section{Corollary: closed subgroup of \texorpdfstring{$\GL_n$}{GL\_n}}

\begin{corollary}[Closed subgroup form]\label{cor:closed-subgroup-of-GL}\issue{13}
\uses{thm:embedding}
For any algebraic group $G$ over a field $k$, there exist a positive
integer $n$ and a closed immersion of group schemes
$G \hookrightarrow \GL_n$ over $\Spec k$.
\end{corollary}
\begin{proof}
Apply the embedding theorem and project the witness
$\GL(W) \cong \GL_n$ for $n = \dim_k W$.
\end{proof}
